# 机试代码模板 · 需求 → 代码

> 机试的实际形态：考官给一句话需求，你在 30~40 分钟内把代码写出来、烧进去、跑通。
> 所以这份文件不放"考察点"，只放**能直接抄的代码**。
>
> 配合 `二面机试库/` 使用（bsp_uart / bsp_motor / bsp_encoder / bsp_key / pid / app_wave / oled）

# 零、通用 main.c 骨架（90% 的题都从这份开始）

```c
/* USER CODE BEGIN Includes */
#include <stdio.h>
#include "bsp_uart.h"
#include "bsp_motor.h"
#include "bsp_encoder.h"
#include "bsp_key.h"
#include "pid.h"
#include "oled.h"
/* USER CODE END Includes */

/* USER CODE BEGIN PV */
static PID_t  g_spd;
volatile int32_t g_target = 0;
volatile float   g_rpm    = 0.0f;
static uint32_t  g_send_t = 0;
/* USER CODE END PV */

/* 中断回调放在最后 */
```

```c
  /* USER CODE BEGIN 2 */
  BSP_UART_Init(&huart1);
  BSP_Motor_Init();
  BSP_Encoder_Init();
  BSP_Key_Init();
  OLED_Init();
  OLED_Clear();

  PID_Init(&g_spd, 0.5f, 0.03f, 0.0f, 450.0f, 3000.0f);

  HAL_TIM_Base_Start_IT(&htim1);              /* 10ms 控制周期 */
  /* USER CODE END 2 */
```

```c
  /* USER CODE BEGIN WHILE */
  while (1)
  {
    /* USER CODE END WHILE */

    /* USER CODE BEGIN 3 */

    /* ---- ① 按键 ---- */
    if (BSP_Key_GetEvent(0)) g_target = 200;
    if (BSP_Key_GetEvent(1)) g_target = 300;

    /* ---- ② OLED ---- */
    OLED_ShowString(0,  0, "T:", 16);
    OLED_ShowNum   (16, 0, (uint32_t)g_target, 4, 16);
    OLED_ShowString(56, 0, "A:", 16);
    OLED_ShowNum   (80, 0, (uint32_t)g_rpm,    4, 16);

    /* ---- ③ 串口 10ms 一次 ---- */
    if (HAL_GetTick() - g_send_t >= 10)
    {
      g_send_t = HAL_GetTick();
      printf("%d,%d\n", (int)g_target, (int)g_rpm);
    }
  }
  /* USER CODE END 3 */
```

```c
/* USER CODE BEGIN 4 */
int _write(int fd, char *ptr, int len)          /* printf 重定向 */
{
  (void)fd;
  HAL_UART_Transmit(&huart1, (uint8_t *)ptr, len, 0xFFFF);
  return len;
}

void HAL_TIM_PeriodElapsedCallback(TIM_HandleTypeDef *htim)
{
  if (htim->Instance != TIM1) return;

  g_rpm = BSP_Encoder_ReadRpm();                 /* 测速 + 位置累计 */

  float u = PID_Calc(&g_spd, (float)g_target, g_rpm);
  BSP_Motor_Set((int16_t)u);
}
/* USER CODE END 4 */
```

**这一份就能应付"PID 定速 + 按键切换 + OLED 显示 + 串口上报"这类题。**

# 一、需求：串口收到什么就回显什么

```c
/* USER CODE BEGIN 2 */
BSP_UART_Init(&huart1);
/* USER CODE END 2 */

/* 主循环里 */
uint8_t buf[64];
uint16_t n = BSP_UART_ReadFrame(buf, sizeof buf);
if (n > 0)
{
    BSP_UART_Send(buf, n);          /* 原样回发 */
}
```

**要点**：`BSP_UART_ReadFrame` 内部靠 IDLE 判断帧结束，不定长也能收。

# 二、需求：串口发 "V200" 设定目标转速

```c
/* 主循环里 */
uint8_t buf[32];
uint16_t n = BSP_UART_ReadFrame(buf, sizeof buf);
if (n >= 2 && buf[0] == 'V')
{
    buf[n] = '\0';
    g_target = atoi((char *)&buf[1]);      /* 需要 #include <stdlib.h> */
}
```

**手写解析版（不想用 atoi）**：

```c
if (n >= 2 && buf[0] == 'V')
{
    int32_t v = 0;
    for (uint16_t i = 1; i < n && buf[i] >= '0' && buf[i] <= '9'; i++)
        v = v * 10 + (buf[i] - '0');
    g_target = v;
}
```

# 三、需求：按键控制占空比，不掉电保存

```c
static int32_t s_duty = 0;      /* 0~450 */

/* 主循环 */
if (BSP_Key_GetEvent(0)) { s_duty += 50; if (s_duty > 450) s_duty = 450; }
if (BSP_Key_GetEvent(1)) { s_duty -= 50; if (s_duty <   0) s_duty =   0; }

BSP_Motor_Set((int16_t)s_duty);

OLED_ShowNum(32, 2, (uint32_t)(s_duty * 100 / 450), 3, 16);   /* 显示百分比 */
```

**注意**：这里不接 PID，`BSP_Motor_Set` 直接当占空比用。

# 四、需求：按一次键电机转 3 圈停住

```c
/* USER CODE BEGIN PV */
static PID_t  g_spd, g_pos;
static int32_t g_target_pulse = 0;
static uint8_t g_running = 0;
/* USER CODE END PV */

/* USER CODE BEGIN 2 */
PID_Init(&g_spd, 0.5f, 0.03f, 0.0f, 450.0f, 3000.0f);
PID_Init(&g_pos, 2.0f, 0.00f, 0.0f, 300.0f, 0.0f);   /* 位置环只用 P */
/* USER CODE END 2 */

/* 主循环 */
if (BSP_Key_GetEvent(0))
{
    BSP_Encoder_ResetPosition();
    g_target_pulse = 3 * (int32_t)ENC_PPR;      /* 3 圈 */
    g_running = 1;
    PID_Reset(&g_pos);
    PID_Reset(&g_spd);
}

if (g_running)
{
    int32_t now_pos = BSP_Encoder_GetPosition();
    if (labs(g_target_pulse - now_pos) < 20)     /* 到位阈值 */
    {
        g_running = 0;
        BSP_Motor_Stop();
        PID_Reset(&g_pos);
        PID_Reset(&g_spd);
    }
}
else
{
    BSP_Motor_Set(0);
}

/* 中断里 */
void HAL_TIM_PeriodElapsedCallback(TIM_HandleTypeDef *htim)
{
  if (htim->Instance != TIM1) return;

  g_rpm = BSP_Encoder_ReadRpm();

  if (!g_running) { BSP_Motor_Stop(); return; }

  /* 位置环输出目标速度（限速） → 速度环输出 PWM */
  float rpm_t = PID_Calc(&g_pos, (float)g_target_pulse,
                          (float)BSP_Encoder_GetPosition());
  if (rpm_t >  300.0f) rpm_t =  300.0f;
  if (rpm_t < -300.0f) rpm_t = -300.0f;

  float u = PID_Calc(&g_spd, rpm_t, g_rpm);
  BSP_Motor_Set((int16_t)u);
}
```

# 五、需求：OLED 显示四个值（目标 / 实际 / 误差 / 占空比）

```c
/* 主循环 */
int32_t err = g_target - (int32_t)g_rpm;

OLED_ShowString(0,  0, "T:", 16);  OLED_ShowNum(16, 0, (uint32_t)g_target, 4, 16);
OLED_ShowString(64, 0, "A:", 16);  OLED_ShowNum(80, 0, (uint32_t)g_rpm,    4, 16);

OLED_ShowString(0,  2, "E:", 16);  OLED_ShowNum(16, 2, (uint32_t)(err < 0 ? -err : err), 4, 16);
OLED_ShowString(64, 2, "D:", 16);  OLED_ShowNum(80, 2, (uint32_t)g_duty,    4, 16);
```

**注意**：`OLED_ShowNum` 只能传 `uint32_t`，负数要先取绝对值再单独显示符号。

# 六、需求：用按键循环切换三种波形

```c
/* USER CODE BEGIN PV */
static WaveMode g_mode = WAVE_STEP;
static float    g_t    = 0.0f;
/* USER CODE END PV */

/* 主循环 */
if (BSP_Key_GetEvent(0))
{
    g_mode = (g_mode == WAVE_STEP)       ? WAVE_SINE_FIX   :
             (g_mode == WAVE_SINE_FIX)   ? WAVE_SINE_SWEEP :
                                           WAVE_STEP;
    g_t = 0.0f;
    PID_Reset(&g_spd);
}

/* 中断里 */
g_t += 0.01f;
if (g_t > 60.0f) g_t = 0.0f;
g_target = APP_Wave_Get(g_mode, g_t);
```

**这就是"直接改枚举切模式"的写法**——三重条件表达式，一行搞定，不需要额外机制。

# 七、机试当天的操作清单

```text
【0~5 分钟】CubeMX
  □ RCC → HSE 晶振，SYS → Serial Wire，HCLK 敲 72
  □ TIM3 PWM（PSC 71 / ARR 999）→ PA6/PA7
  □ TIM2 Encoder（PSC 0 / ARR 65535 / TI1 and TI2）→ PA0/PA1 上拉
  □ TIM1 时基（PSC 7199 / ARR 99）+ NVIC update 中断
  □ USART1 115200 + DMA(TX) + 全局中断
  □ I2C1 → PB6/PB7
  □ 按键 GPIO_EXTIx 下降沿 + 上拉 + NVIC
  □ PA3 GPIO_Output 初始 High（STBY）

【5~8 分钟】拷文件
  □ 7 个 .c 到 Core/Src，8 个 .h 到 Core/Inc
  □ CMakeLists 里 target_sources 加上

【8~12 分钟】抄 main 模板（上面的"零"）
【12~15 分钟】编译烧录，确认四条链路
  □ 串口有输出    □ OLED 亮    □ 电机能转    □ 按键有反应
【15~35 分钟】写业务逻辑
```

# 八、卡住时的逃生顺序

```text
1. 串口没输出      → 检查 PA9/PA10 接线、printf 有没有 \n
2. OLED 不亮       → 检查 I2C 地址 0x78、上拉、PA6/PA7 有没有冲突
3. 电机不转        → STBY 高？PWMA 高？12V 有？CCR 值对不对
4. 编码器不动      → HAL_TIM_Encoder_Start 传 ALL 了吗？PA0/PA1 上拉了吗
5. PID 跑飞        → target 是不是 0？方向反了吗？integral 爆了吗
6. 编译过但没反应  → 启动代码有没有放在 MX_xxx_Init 之前（会 HardFault）
```

**每一条都是 30 秒内能查完的，别在一个问题上耗超过 3 分钟。**
